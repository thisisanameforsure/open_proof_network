```python
"""F07-T40 mutants, chosen by OPN_MUTANT; each weakens one rule of on_the_node in memory."""

import os


def pytest_configure(config):
    from opn_api import submissions

    which = os.environ["OPN_MUTANT"]
    original = submissions.on_the_node

    if which == "no-rivals":
        def on_the_node(ctx, node_id, artifact_type, *, proved, tutorial):
            out = original(ctx, node_id, artifact_type, proved=proved, tutorial=tutorial)
            out.pop("rivals", None)
            return out
    elif which == "rivals-ignore-blocking":
        def on_the_node(ctx, node_id, artifact_type, *, proved, tutorial):
            out = original(ctx, node_id, artifact_type, proved=proved, tutorial=tutorial)
            every = [
                {"pr_number": f.pr_number, "pseudonym": f.pseudonym}
                for f in ctx.store.list_open_submissions()
                if f.node_id == node_id and f.kind in {"proof", "partial"}
            ]
            if every:
                out["rivals"] = every
            return out
    elif which == "empty-keys":
        def on_the_node(ctx, node_id, artifact_type, *, proved, tutorial):
            out = original(ctx, node_id, artifact_type, proved=proved, tutorial=tutorial)
            return {"rivals": [], "node_proved": False, **out}
    elif which == "never-proved":
        def on_the_node(ctx, node_id, artifact_type, *, proved, tutorial):
            return original(ctx, node_id, artifact_type, proved=False, tutorial=tutorial)
    elif which == "becomes-for-every-type":
        def on_the_node(ctx, node_id, artifact_type, *, proved, tutorial):
            return original(ctx, node_id, "proof", proved=proved, tutorial=tutorial)
    else:
        raise SystemExit(which)
    submissions.on_the_node = on_the_node
```
