"""Mutants for F10-T50's tests (pytest -p mutant_lit; OPN_MUTANT=A|B). Nothing on disk changes."""
import os
import yaml
from opn_gate import context

WHICH = os.environ["OPN_MUTANT"]
real = context._literature_inline

def uncapped(reader, node_dir, lit):  # A: no caps, no fallback
    if lit is None:
        return None
    out = dict(lit)
    doc = yaml.safe_load(reader.read(f"{node_dir}/{lit['record']}").decode("utf-8"))
    out["summary"] = doc["summary"]
    out["references"] = doc["references"]
    return out

def disk_only(reader, node_dir, lit):  # B: the gate inlines, a host reader does not
    if isinstance(reader, context.DiskReader):
        return real(reader, node_dir, lit)
    return None if lit is None else dict(lit)

context._literature_inline = uncapped if WHICH == "A" else disk_only
