"""F08-T18 mutants, chosen by OPN_MUTANT; each weakens one rule in memory, no file edited."""

import os


def pytest_configure(config):
    from opn_api import claims, duplicates, requests, store
    from opn_gate import graph as graphmod

    which = os.environ["OPN_MUTANT"]
    original = requests.known_state

    def rivals(ctx, stmt_ref, keep):
        return [
            {"pr_number": f.pr_number, "pseudonym": f.pseudonym}
            for f in duplicates.open_rivals(ctx, stmt_ref, frozenset({"defect-claim"}))
            if keep(f)
        ]

    if which == "never-refuses":
        def known_state(ctx, defect_class, stmt_ref, facts):
            if facts is None:
                return []
            return rivals(ctx, stmt_ref, lambda f: f.defect_class == defect_class)
    elif which == "refuses-every-class":
        def known_state(ctx, defect_class, stmt_ref, facts):
            if facts is not None and facts.get("cause") == graphmod.CAUSE_CIRCULAR:
                raise claims.circular(stmt_ref, facts)
            return original(ctx, defect_class, stmt_ref, facts)
    elif which == "names-every-class":
        def known_state(ctx, defect_class, stmt_ref, facts):
            original(ctx, defect_class, stmt_ref, facts)
            return rivals(ctx, stmt_ref, lambda f: True)
    elif which == "names-nothing":
        def known_state(ctx, defect_class, stmt_ref, facts):
            original(ctx, defect_class, stmt_ref, facts)
            return []
    elif which == "no-claim-paths":
        requests.merged_claims_of = lambda *a, **k: ()
        return
    elif which == "store-drops-class":
        old = store._submission

        def _submission(record):
            import dataclasses

            return dataclasses.replace(old(record), defect_class=None)

        store._submission = _submission
        return
    else:
        raise SystemExit(which)
    requests.known_state = known_state
