```python
"""F08-T19 mutants, chosen by OPN_MUTANT; each weakens one rule in memory, no file edited."""

import os
import types


def pytest_configure(config):
    from opn_api import duplicates, proposals
    from opn_gate import admit

    which = os.environ["OPN_MUTANT"]
    if which == "open-proposals-ignored":
        proposals.proposed_statements = lambda ctx, target_id, found: iter(())
    elif which == "open-ignores-blocking":
        original = proposals.proposed_statements

        def proposed_statements(ctx, target_id, found):
            saved = duplicates.blocks
            duplicates.blocks = lambda ctx, f: True
            try:
                yield from original(ctx, target_id, found)
            finally:
                duplicates.blocks = saved

        proposals.proposed_statements = proposed_statements
    elif which == "open-includes-own-node":
        original_check = proposals.check_declaration_free
        proposals.check_declaration_free = lambda ctx, t, s, n: original_check(ctx, t, s, "\0")
    elif which == "supersedes-ignored":
        original_holder = admit.declaration_holder

        def declaration_holder(decl_name, siblings, *, own=None, supersedes=None):
            return original_holder(decl_name, siblings, own=own)

        admit.declaration_holder = declaration_holder
    elif which == "service-keeps-a-copy":
        # the service's own copy of the gate's function and words, as before F08-T19
        proposals.admit = types.SimpleNamespace(
            declaration_holder=admit.declaration_holder,
            clash_message=lambda holder, decl: (
                f"node {holder!r} already declares {decl}; a node may restate a declaration "
                "only by superseding the node that holds it, D-8"
            ),
        )
    else:
        raise SystemExit(which)
```
