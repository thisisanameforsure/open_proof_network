import json, sys, pathlib, mcp
target, node, path, mode = sys.argv[1], sys.argv[2], sys.argv[3], (sys.argv[4] if len(sys.argv)>4 else "verify")
content = pathlib.Path(path).read_text(encoding="utf-8")
args = {"target_id": target, "content": content, "mode": mode}
if node != "-": args["node_id"] = node
st, r = mcp.call('tools/call', {'name':'check_lean','arguments':args})
c = r['result'].get('structuredContent', r['result'])
pathlib.Path("last-check.json").write_text(json.dumps(c, indent=2))
print("http", st)
print(json.dumps(c, indent=2)[:4000])
