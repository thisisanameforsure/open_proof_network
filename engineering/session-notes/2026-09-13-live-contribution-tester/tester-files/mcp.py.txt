#!/usr/bin/env python3
"""Call one MCP tool on the live endpoint: mcp.py <tool> '<json args>' [--token T] [--raw]"""
import json, sys, urllib.request, os

URL = "https://api.openproofnetwork.org/mcp"

def call(tool, args, token=None):
    body = json.dumps({"jsonrpc": "2.0", "id": 1, "method": "tools/call",
                       "params": {"name": tool, "arguments": args}}).encode()
    req = urllib.request.Request(URL, data=body, method="POST")
    req.add_header("Content-Type", "application/json")
    req.add_header("Accept", "application/json, text/event-stream")
    if token:
        req.add_header("Authorization", "Bearer " + token)
    try:
        with urllib.request.urlopen(req, timeout=120) as r:
            return r.status, dict(r.headers), r.read().decode()
    except urllib.error.HTTPError as e:
        return e.code, dict(e.headers), e.read().decode()

if __name__ == "__main__":
    tool = sys.argv[1]
    args = json.loads(sys.argv[2]) if len(sys.argv) > 2 and not sys.argv[2].startswith("--") else {}
    token = None
    if "--token" in sys.argv:
        token = sys.argv[sys.argv.index("--token") + 1]
    status, headers, text = call(tool, args, token)
    print(f"[HTTP {status}]", file=sys.stderr)
    if "--raw" in sys.argv:
        print(text)
    else:
        try:
            d = json.loads(text)
            r = d.get("result", d)
            if isinstance(r, dict) and "structuredContent" in r:
                print(json.dumps(r["structuredContent"], indent=1, ensure_ascii=False))
                if r.get("isError"):
                    print("[isError true]", file=sys.stderr)
            else:
                print(json.dumps(d, indent=1, ensure_ascii=False))
        except Exception:
            print(text)
