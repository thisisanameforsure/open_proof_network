# ---- status glyphs: shape carries meaning, colour repeats it
def glyph(kind, size=12):
    if kind=='proved': st=f'background: {PROVED}; border: 1.5px solid {PROVED}'
    elif kind=='failed': st=f'background: linear-gradient(45deg, transparent 43%, {FAIL} 43%, {FAIL} 57%, transparent 57%); border: 1.5px solid {FAIL}'
    else: st=f'background: transparent; border: 1.5px solid {INK}'
    return f'<span aria-hidden="true" style="flex: none; display: inline-block; width: {size}px; height: {size}px; border-radius: 50%; box-sizing: border-box; {st}"></span>'
def status(kind, label=None):
    label=label or {'proved':'proved','open':'open','failed':'refuted'}[kind]
    col={'proved':PROVED,'failed':FAIL}.get(kind,INK)
    return f'<span style="display: inline-flex; align-items: center; gap: 6px; font-family: {SANS}; font-size: 14px; color: {col}; white-space: nowrap">{glyph(kind)}{label}</span>'

def stages(done=0, compact=False):
    """The signature mark: proved → explained → written up. done = how many are reached."""
    names=['proved','explained','written up']
    parts=[]
    for i,n in enumerate(names):
        on=i<done
        dot=f'<span aria-hidden="true" style="flex: none; width: 10px; height: 10px; border-radius: 50%; box-sizing: border-box; border: 1.5px solid {PROVED if on else "#9A968D"}; background: {PROVED if on else "transparent"}"></span>'
        lab='' if compact else f'<span style="color: {INK if on else MUTED}">{n}</span>'
        parts.append(f'<span style="display: inline-flex; align-items: center; gap: 6px">{dot}{lab}</span>')
        if i<2: parts.append(f'<span aria-hidden="true" style="width: {10 if compact else 16}px; height: 1.5px; background: {PROVED if i+1<done else "#C9C5BC"}"></span>')
    aria=f'{done} of 3: ' + ', '.join(f"{n} {'yes' if i<done else 'not yet'}" for i,n in enumerate(names))
    return f'<span role="img" aria-label="{aria}" style="display: inline-flex; align-items: center; gap: 6px; font-family: {SANS}; font-size: 13px; white-space: nowrap">{"".join(parts)}</span>'

# ---- figure drawing: absolutely placed nodes and straight edges (no single <svg>)
import math as _m
def edge(x1,y1,x2,y2,color='#B9B5AC',w=1.25,dash=False,op=1):
    L=_m.hypot(x2-x1,y2-y1); a=_m.degrees(_m.atan2(y2-y1,x2-x1))
    st=f'border-top: {w}px dashed {color}; height: 0' if dash else f'height: {w}px; background: {color}'
    return f'<span aria-hidden="true" style="position: absolute; left: {x1:.1f}px; top: {y1:.1f}px; width: {L:.1f}px; {st}; transform: rotate({a:.2f}deg); transform-origin: 0 50%; opacity: {op}"></span>'
def node(x,y,label,kind='open',sel=False,op=1,size=12.5,mono=False,maxw=None):
    ff=MONO if mono else SANS
    bd=f'2px solid {INK}' if sel else f'1px solid {"#CFCBC2" if op<1 else INK if kind=="root" else "#CFCBC2"}'
    mw=f'width: max-content; max-width: {maxw}px; white-space: normal;' if maxw else 'white-space: nowrap;'
    g=glyph('open' if kind=='root' else kind,10)
    bgc, fgc = (INK, '#FFFFFF') if sel else (SURF, INK)
    if sel: g=g.replace(f'border: 1.5px solid {INK}','border: 1.5px solid #FFFFFF')
    return f'<a href="#" style="position: absolute; left: {x}px; top: {y}px; transform: translate(-50%, -50%); display: inline-flex; gap: 7px; align-items: center; padding: 6px 10px; background: {bgc}; border: {bd}; border-radius: 4px; text-decoration: none; color: {fgc}; font-family: {ff}; font-size: {size}px; line-height: 1.3; {mw} opacity: {op}">{g}<span>{label}</span></a>'
def junction(x,y,op=1,kind='all'):
    if kind=='any': return f'<span aria-hidden="true" style="position: absolute; left: {x-5}px; top: {y-5}px; width: 10px; height: 10px; background: {SURF}; border: 1.5px solid {INK}; transform: rotate(45deg); opacity: {op}"></span>'
    return f'<span aria-hidden="true" style="position: absolute; left: {x-4}px; top: {y-4}px; width: 8px; height: 8px; border-radius: 50%; background: {INK}; opacity: {op}"></span>'
def canvas_box(inner,w,h,label):
    return f'<div role="group" aria-label="{label}" style="position: relative; width: {w}px; height: {h}px; flex: none">{inner}</div>'
