def T(label,kind='open',ident='',children=None,uses=None,split=None,spec=None):
    return dict(label=label,kind=kind,id=ident,children=children or [],uses=uses,split=split,spec=spec)
def _leaves(n):
    return 1 if not n['children'] else sum(_leaves(c) for c in n['children'])
def draw(tree, sel_id=None, path=()):
    items=[]; edges=[]; maxd=[0]; pos={}
    def place(n,x0,d):
        w=_leaves(n)*GAP; x=x0+w/2; y=40+d*DY; maxd[0]=max(maxd[0],d)
        if n['children']:
            cx=x0; jy=y+DY*0.5
            edges.append(edge(x,y+16,x,jy,color=INK,w=2.25 if n['id'] in path else 1.25))
            for c in n['children']:
                cw=_leaves(c)*GAP; ccx=cx+cw/2
                onp=c['id'] in path
                edges.append(edge(x,jy,ccx,y+DY-16,color=INK if onp else '#8F8B82',w=2.25 if onp else 1.1))
                place(c,cx,d+1); cx+=cw
            items.append(junction(x,jy))
            if n.get('split'): items.append(f'<span style="position: absolute; left: {x-10}px; top: {jy-9}px; transform: translateX(-100%); font-family: {SANS}; font-size: 11.5px; color: {MUTED}; white-space: nowrap; background: {SURF}; padding: 0 2px">{n["split"]}</span>')
        k=n['kind']
        if k=='fold':
            items.append(f'<button type="button" style="position: absolute; left: {x}px; top: {y}px; transform: translate(-50%,-50%); font-family: {SANS}; font-size: 12.5px; padding: 7px 11px; border: 1px dashed {PROVED}; border-radius: 4px; background: {PROVED_SOFT}; color: {PROVED}; cursor: pointer; white-space: nowrap; width: max-content; text-align: left; line-height: 1.3">{n["label"]}</button>')
        else:
            items.append(node(x,y,n['label'],'root' if k=='root' else k,sel=(n['id']==sel_id),maxw=140,size=12.5))
        if n.get('spec'):
            right=bool(n['children'])
            sx=x+82 if right else x; sy=y-11 if right else y+30
            items.append(f'<a href="#spec" style="position: absolute; left: {sx}px; top: {sy}px; transform: {"none" if right else "translateX(-50%)"}; font-family: {SANS}; font-size: 11.5px; color: {MUTED}; background: {SURF}; border: 1px dotted #8F8B82; border-radius: 4px; padding: 3px 7px; white-space: nowrap; text-decoration: none">+ {n["spec"]} ↓</a>')
        if n.get('uses'):
            ux=x+GAP*0.42; uy=y+DY*0.55
            edges.append(edge(x+40,y+12,ux-30,uy-12,color='#8F8B82',dash=True))
            items.append(f'<span style="position: absolute; left: {ux}px; top: {uy}px; transform: translate(-50%,-50%); font-family: {SANS}; font-size: 11.5px; color: {MUTED}; background: {BG}; border: 1px dashed #B9B5AC; border-radius: 4px; padding: 4px 8px; white-space: nowrap">uses: {n["uses"]}</span>')
    place(tree,0,0)
    W=_leaves(tree)*GAP; H=40+maxd[0]*DY+44
    pos_out[:] = []
    return canvas_box(''.join(edges)+''.join(items), W, H, 'Route drawn as a tree')
