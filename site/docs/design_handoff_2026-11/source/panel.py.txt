def mono(t,sz='0.85em'): return f'<code style="font-family: {MONO}; font-size: {sz}">{t}</code>'
def sub(t): return f'<p style="margin: 0; font-family: {SANS}; font-size: 12px; font-weight: 500; letter-spacing: 0.08em; text-transform: uppercase; color: {MUTED}">{t}</p>'
def fold(summary, body):
    return f'<details style="border-top: 1px solid {LINE}; padding: 10px 0 0"><summary style="cursor: pointer; font-family: {SANS}; font-size: 14px">{summary}</summary><div style="padding-top: 8px; display: flex; flex-direction: column; gap: 6px">{body}</div></details>'
def facts(rows):
    return '<dl style="margin: 0; display: grid; grid-template-columns: repeat(3, minmax(0, 1fr)); gap: 6px 12px; font-family: '+SANS+'; font-size: 13.5px">'+''.join(f'<dt style="grid-column: span 1; color: {MUTED}">{a}</dt><dd style="grid-column: span 2; margin: 0; overflow-wrap: anywhere">{b}</dd>' for a,b in rows)+'</dl>'
def holes(items):
    return '<ul style="list-style: none; margin: 0; padding: 0; display: flex; flex-direction: column; gap: 6px; font-family: '+SANS+'; font-size: 13.5px">'+''.join(f'<li style="display: flex; gap: 8px; align-items: flex-start">{glyph(k,10)}<span style="margin-top: -3px"><a href="#">{l}</a>{(" " + mono(n,"11.5px")) if n else ""}</span></li>' for k,l,n in items)+'</ul>'
